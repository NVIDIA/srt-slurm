## service_gms_agg_0_node-01
nodelist: node-01
output: <output>/logs/service_gms_agg_0_node-01.out
container_image: dynamo-vllm
container_mounts:
    <repo>/src/srtctl/benchmarks/scripts: /srtctl-benchmarks
    <repo>/src/srtctl/runtime_scripts: /srtctl-runtime
    <model>: /model
    <output>/logs: /logs
env_to_set:
    CUDA_VISIBLE_DEVICES: 0
    ETCD_ENDPOINTS: http://127.0.0.1:2379
    GMS_SOCKET_DIR: /dev/shm/srtctl-4242/agg_0
use_bash_wrapper: True
command: |
    bash -c 'set -u
    dir=/dev/shm/srtctl-4242/agg_0
    mkdir -p "$dir" && rm -f "$dir"/gms_*.sock || exit 1
    export GMS_SOCKET_DIR="$dir"
    pids=()
    for device in $(seq 0 0); do
        python3 -m gpu_memory_service --device "$device" &
        pids+=($!)
    done
    stopping=0
    stop() { stopping=1; kill -TERM "${pids[@]}" 2>/dev/null; }
    trap stop TERM INT
    count=0
    for _ in $(seq 1 120); do
        for pid in "${pids[@]}"; do
            if ! kill -0 "$pid" 2>/dev/null; then
                echo "GMS server pid $pid exited during startup" >&2
                stop; wait; exit 1
            fi
        done
        count=$(ls "$dir"/gms_*.sock 2>/dev/null | wc -l)
        if [ "$count" -ge 2 ]; then break; fi
        sleep 1
    done
    if [ "$count" -lt 2 ]; then
        echo "GMS startup timed out: $count of 2 sockets in $dir after 120s" >&2
        stop; wait; exit 1
    fi
    echo "GMS ready: 1 device(s), $count sockets in $dir"
    wait -n "${pids[@]}"
    rc=$?
    if [ "$stopping" = 1 ]; then
        wait
        rm -rf -- "$dir"
        exit 0
    fi
    echo "GMS server exited (code=$rc); stopping the rest" >&2
    stop; wait
    if [ "$rc" = 0 ]; then rc=1; fi
    exit "$rc"
    '
