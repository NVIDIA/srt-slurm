nodelist: node-01
output: <output>/logs/service_gateway.out
container_image: sglang
container_mounts:
    <repo>/src/srtctl/benchmarks/scripts: /srtctl-benchmarks
    <repo>/src/srtctl/runtime_scripts: /srtctl-runtime
    <model>: /model
    <output>/logs: /logs
env_to_set:
    GATEWAY_PORT: 18400
    LLM_API_BASE: http://127.0.0.1:8000
use_bash_wrapper: True
command: |
    python3 -c 'import http.server, os, urllib.error, urllib.request
    upstream, count = os.environ["LLM_API_BASE"].rstrip("/"), 0
    class Proxy(http.server.BaseHTTPRequestHandler):
        def _forward(self):
            global count
            if self.path == "/healthz":
                self.send_response(200); self.end_headers(); self.wfile.write(b"ok"); return
            if self.path == "/metrics":
                body = f"# TYPE gateway_requests_total counter\ngateway_requests_total {count}\n".encode()
                self.send_response(200); self.send_header("Content-Type", "text/plain"); self.end_headers(); self.wfile.write(body); return
            count += 1
            length = int(self.headers.get("Content-Length", "0"))
            data = self.rfile.read(length) if length else None
            hdrs = {k: v for k, v in self.headers.items() if k.lower() not in ("host", "content-length")}
            req = urllib.request.Request(upstream + self.path, data=data, headers=hdrs, method=self.command)
            try:
                with urllib.request.urlopen(req, timeout=300) as r:
                    body = r.read(); self.send_response(r.status); self.send_header("Content-Type", r.headers.get("Content-Type", "application/json")); self.end_headers(); self.wfile.write(body)
            except urllib.error.HTTPError as e:
                body = e.read(); self.send_response(e.code); self.end_headers(); self.wfile.write(body)
        do_GET = do_POST = _forward
        def log_message(self, *args): pass
    http.server.ThreadingHTTPServer(("0.0.0.0", int(os.environ["GATEWAY_PORT"])), Proxy).serve_forever()
    '
